-- Cấu hình Key và thời gian lưu (24 giờ = 86400 giây)
local CORRECT_KEY = "ChilliHubKeyHdiehBskaowoejVdjsowieBdkaowiJsjwowop"
local SCRIPT_URL = "https://raw.githubusercontent.com/tienkhanh1/spicy/main/Chilli.lua"
local KEY_FILE = "ChilliHub_SavedKey.json"
local KEY_DURATION = 86400 
local TweenService = game:GetService("TweenService")

-- Hàm chạy script chính kèm bộ bắt lỗi an toàn
local function loadMainScript()
    local success, err = pcall(function()
        loadstring(game:HttpGet(SCRIPT_URL))()
    end)
    if not success then
        warn("Chilli Hub Script Error: " .. tostring(err))
    end
end

-- Hàm kiểm tra xem đã lưu key hợp lệ trong 24h chưa
local function checkSavedKey()
    if readfile and isfile and isfile(KEY_FILE) then
        local success, data = pcall(function()
            return game:GetService("HttpService"):JSONDecode(readfile(KEY_FILE))
        end)
        if success and data and data.Key == CORRECT_KEY then
            local currentTime = os.time()
            if (currentTime - data.Timestamp) < KEY_DURATION then
                return true -- Key vẫn còn hạn, tự động bỏ qua bảng key
            end
        end
    end
    return false
end

-- Hàm lưu key mới
local function saveKey()
    if writefile then
        local data = {
            Key = CORRECT_KEY,
            Timestamp = os.time()
        }
        pcall(function()
            writefile(KEY_FILE, game:GetService("HttpService"):JSONEncode(data))
        end)
    end
end

-- Nếu đã lưu key hợp lệ từ trước, tự động chạy luôn script mà không hiện bảng Key
if checkSavedKey() then
    loadMainScript()
    return
end

local function smoothShake()
    local originalPos = UDim2.new(0.5, -200, 0.5, -135)
    local tweenInfo = TweenInfo.new(0.05, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
    
    local shakeOffsets = {-8, 8, -6, 6, -3, 3, 0}
    for _, offset in ipairs(shakeOffsets) do
        local tween = TweenService:Create(MainFrame, tweenInfo, {Position = originalPos + UDim2.new(0, offset, 0, 0)})
        tween:Play()
        tween.Completed:Wait()
    end
end
